เชื่อมต่อ Google Drive จริง (ผ่าน Google Apps Script)

ซิงค์ภาพถ่ายจากโฟลเดอร์ Google Drive แบบ Real-time

* หากเว้นว่างไว้ ระบบจะทำงานใน โหมดสาธิต (Demo Mode) พร้อมภาพตัวอย่างให้ทดสอบทันที

2. ขั้นตอนสร้างตัวเชื่อมต่อใน Google Drive (ใช้เวลา 2 นาที)

  1. ไปที่ script.google.com แล้วกด New Project (โครงการใหม่)
  2. ลบโค้ดเดิมออกทั้งหมด แล้วคัดลอกโค้ด Code.gs ด้านล่างไปวางแทนที่
  3. กดปุ่ม Deploy (การทำให้ใช้งานได้) > New deployment (การทำให้ใช้งานได้รายการใหม่)
  4. เลือกประเภทเป็น Web app, ตั้งค่า Who has access เป็น Anyone (ทุกคน) แล้วกด Deploy
  5. นำ Web app URL ที่ได้มาวางในช่องด้านบน และตั้งค่าแชร์โฟลเดอร์ Google Drive เป็น "ทุกคนที่มีลิงก์สามารถดูได้"
Code.gs (Google Apps Script Backend) 
function doGet(e) {
  var folderId = e.parameter.folderId;
  if (!folderId) {
    return ContentService.createTextOutput(JSON.stringify({ error: "Missing folderId" }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  
  var folder = DriveApp.getFolderById(folderId);
  var files = folder.getFiles();
  var result = [];
  
  while (files.hasNext()) {
    var file = files.next();
    var mime = file.getMimeType();
    if (mime.indexOf("image/") === 0) {
      var id = file.getId();
      result.push({
        id: id,
        name: file.getName(),
        timestamp: file.getDateCreated().getTime(),
        size: (file.getSize() / (1024 * 1024)).toFixed(1) + " MB",
        thumbUrl: "https://drive.google.com/thumbnail?id=" + id + "&sz=w800",
        fullUrl: "https://drive.google.com/thumbnail?id=" + id + "&sz=w2048",
        downloadUrl: "https://drive.google.com/uc?export=download&id=" + id
      });
    }
  }
  
  result.sort(function(a, b) { return b.timestamp - a.timestamp; });
  
  return ContentService.createTextOutput(JSON.stringify({ photos: result }))
    .setMimeType(ContentService.MimeType.JSON);
}
